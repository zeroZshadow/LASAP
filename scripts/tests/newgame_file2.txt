# From a fresh boot to a new game in file 2. Restore the never-played file 2 first: reset_saves.sh newgame. Each press
# waits until the screen takes it (TESTING.md, "Game state") and stops the script within seconds when it is not
# taken.
connect 90
# The intro ends on a button once the title exists; the title takes L+R in WaitLR.
tapuntil 1 screen=title when title:Deactive 30
tapuntil c0 title:GameStart when title:WaitLR+unlocked 20
# File 2 is slot 1 (the cursor starts there, on the last file used). A opens it, A again starts it.
select 1
tapuntil 1 files:DataSelectConfirm when files:DataSelectKeyWait
tapuntil 1 files:GameStart when files:DataSelectConfirm
# Link wakes up in Tarin's house; Marin's lines close with B, pressed only while a line waits for it.
inlevel 60
level MarinTarinHouse
