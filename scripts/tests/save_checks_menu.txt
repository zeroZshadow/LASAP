# Check files by save (DESIGN.md, Persistence), part 2 of 2, on a fresh boot after save_checks_play: file 2 has its
# own check file with location 7, which the autosave keeps too (it holds file 2's save). A copy of file 2 to file 3
# gets an id and a check file of its own; erasing the copy deletes its file; continuing a second copy sends
# location 7 from the copy's file.
connect 90
tapuntil 1 screen=title when title:Deactive 30
tapuntil c0 title:GameStart when title:WaitLR+unlocked 20
until files:DataSelectKeyWait 20
send DBG checkfiles
expect ^CHECKFILES\s[0-9a-f]{32}\.chk$ 5

# Copy file 2 (Start, then Copy below it) to the empty file 3, the first target; the confirmation starts on Cancel
select 1
tapuntil 1 files:DataSelectConfirm when files:DataSelectKeyWait
tap 8000 when files:DataSelectConfirm
tapuntil 1 files:CopyDataSelect when files:DataSelectConfirm 5
tapuntil 1 files:CopyDataConfirm when files:CopyDataSelect 5
tap 4000 when files:CopyDataConfirm
tapuntil 1 !files:CopyDataConfirm when files:CopyDataConfirm 5
until files:CopyDataEnd 15
send DBG checkfiles
expect ^CHECKFILES\s[0-9a-f]{32}\.chk\s[0-9a-f]{32}\.chk$ 5
tapuntil 1 files:DataSelectKeyWait when files:CopyDataEnd 10

# Erase the copy (Delete right of Copy): X+L+R held
select 2
tapuntil 1 files:DataSelectConfirm when files:DataSelectKeyWait
tap 8000 when files:DataSelectConfirm
tap 4000 when files:DataSelectConfirm
tapuntil 1 files:DeleteDataWaitKey when files:DataSelectConfirm 5
hold c4 0 0 until files:DeleteDataEnd 10
send DBG checkfiles
expect ^CHECKFILES\s[0-9a-f]{32}\.chk$ 5
tapuntil 1 files:DataSelectKeyWait when files:DeleteDataEnd 10

# Copy again and continue the copy: its checks come from its own file
select 1
tapuntil 1 files:DataSelectConfirm when files:DataSelectKeyWait
tap 8000 when files:DataSelectConfirm
tapuntil 1 files:CopyDataSelect when files:DataSelectConfirm 5
tapuntil 1 files:CopyDataConfirm when files:CopyDataSelect 5
tap 4000 when files:CopyDataConfirm
tapuntil 1 !files:CopyDataConfirm when files:CopyDataConfirm 5
until files:CopyDataEnd 15
tapuntil 1 files:DataSelectKeyWait when files:CopyDataEnd 10
select 2
tapuntil 1 files:DataSelectConfirm when files:DataSelectKeyWait
tapuntil 1 files:GameStart when files:DataSelectConfirm
inlevel 60
expect CHECKS\s7$ 20
send DBG checkfiles
expect ^CHECKFILES\s[0-9a-f]{32}\.chk\s[0-9a-f]{32}\.chk$ 5
