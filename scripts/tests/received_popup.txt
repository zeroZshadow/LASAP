# Instant Pickups on Incoming or Everything (a mod patched with it, or config.ini [tweaks] instant_pickups): a received item
# shows its model above Link's head and the toast names it, while Link keeps walking; no event starts. Then a song,
# whose model is Marin (an NPC model, sized and centred as in the item-get), and a trap, which still plays its effect
# as an event, and a level load while a popup shows. Run after newgame_file2; starts on the open lawn east of Mabe's
# phone booth tree (Field_12D).
# Screenshots: popup_item, popup_song.
# With the sword and the shield Link's walk is anim=walk (idle_bare without them).
send DBG give 0 1 0
expect OK.give 5
send DBG give 2 1 0
expect OK.give 5
warp Field 43 10.5 137
guard hurt
free 30
guard event=22
send DBG grant Seashell 1 -1
expect OK.grant 5
await 3 popup=1
await 3 shown=22(\s|$)
# The toast names the item
send DBG text SystemSaving
expect said=You\sgot\sSeashell 5
# Link walks while the model shows
hold 0 0 20000 until anim=walk 3
await 1 popup=1
shot popup_item.png
await 5 popup=0
check event=0\s.*showq=0(\s|$)
send DBG grant Song_WindFish 1 -1
expect OK.grant 5
await 3 popup=1
wait 0.6
shot popup_song.png
await 5 popup=0
# A received instrument plays its song
send DBG grant WindMarimba 1 -1
expect OK.grant 5
await 3 popup=1
await 3 bgm=\S*BGM_INST_MARIMBA:2
await 5 popup=0
guard off event=22
guard off hurt
# A level load during a popup takes the model with Link; the next popup shows in the new level
send DBG grant Seashell 1 -1
expect OK.grant 5
await 3 popup=1
warp Field 38.25 10.5 128.25
check popup=0
send DBG grant Seashell 1 -1
expect OK.grant 5
await 3 popup=1
await 5 popup=0
# A trap plays its effect as an event
send DBG grant ZapTrap 1 -1
expect OK.grant 5
await 5 event=22
check popup=0
free 20
