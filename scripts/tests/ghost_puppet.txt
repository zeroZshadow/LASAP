# Multiplayer ghosts, the puppets (DESIGN.md, Ghosts). Run after boot_file1, set up as ghost_net (a ghost relay, the
# client joined to it with --ghost-secret test, scripts/ghost_peer.py --secret test circling), on a mod with
# ghost_puppet.py install. On the open lawn east of Mabe's phone booth tree (Field_12D).
connect
expect HELLO 30
warp Field 43 10.5 137
free 10
# the peer beside Link: ghost slot 0 gets a puppet with Link's model (an NpcGhost with the APGhost row; the
# NpcGhost row draws the ammo ghost)
peer goto Field 46.5 10.5 137
await 5 ghosts=1
await 5 puppets=1
send DBG puppet
expect \|\s0\sactor=\w+\sname=NpcGhost:\w+\spos=[\d.,]+\smodel=[0-9a-f]+
# its name tag over its head, with the name's glyphs drawn
send DBG puppet tags on
await 5 tags=d---
# d-pad up turns the name tags off and on again; the client keeps the setting
tap 2000 when free
await 5 tags=h---
logged Ghost\sname\stags\sare\soff 5 client
tap 2000 when free
await 5 tags=d---
logged Ghost\sname\stags\sare\son 5 client
# the peer leaves the level: the puppet hides; it comes back: the same actor shows again
peer nowhere
await 5 puppets=0
await 5 tags=h---
send DBG puppet
expect \|\s0\sactor=\w+\shidden
peer goto Field 46.5 10.5 137
await 5 puppets=1
send DBG puppet
expect \|\s0\sactor=\w+\sname=\S+\spos=[\d.,]+\smodel=[0-9a-f]+
# Link reloads the level with the ghost shown: the puppet goes with the old level, a new one shows
warp Field 43 10.5 137
free 10
await 10 puppets=1
await 5 tags=d---
# both in a house: the same actor shows there, with the model it loads in that level
peer goto MarinTarinHouse 9 0 10
warp MarinTarinHouse 7.5 0 10
free 10
await 10 puppets=1
send DBG puppet
expect \|\s0\sactor=\w+\sname=\S+\spos=(7|8|9|10)\.\d+,0\.\d+,(8|9|10|11)\.\d+\smodel=[0-9a-f]+
peer goto Field 46.5 10.5 137
warp Field 43 10.5 137
free 10
await 10 puppets=1
# ghosts off in the client: hidden, and after 10 s removed
send /ghosts off
await 5 puppets=0/1
await 15 puppets=0/0
send /ghosts on
await 10 puppets=1/1
# the peer's look: the puppet draws its shapes and tunic (red, the mirror shield and the second sword, the ocarina)
peer shapes ccbfa
peer tunic 1
await 5 gear=ccbfa/1/
peer shapes 32bf8
peer tunic 0
await 5 gear=32bf8/0/
# the peer's face (the item-get's) and upper body (carrying): on the puppet's face slots and upper layer
peer face 3 3 0
peer upper 38
await 5 gear=32bf8/0/3,3,0/38\s
peer face 0 0 0
peer upper 0
await 5 gear=32bf8/0/0,0,0/0\s
# the ghost does not block Link: the peer stands in his way, he runs through it (blocked, he would stop near 44.2)
peer radius 0
peer goto Field 44.6 10.5 137
logged RADIUS\s0 5 peer
logged AT\sField\s44\.6 5 peer
send DBG input 0 700 32767 0
await 3 anim=run
await 3 anim=wait
at 46 137 1.2
peer radius 1.5
peer goto Field 46.5 10.5 137
# Link's own animations, sent by id, on DBG puppet's puppet: his path 20 frames late, replayed as a ghost's lines are
send DBG puppet spawn 179 0 -2
send DBG puppet follow 20
send DBG puppet steer 20
send DBG puppet animlog
send DBG input 0 1200 32767 0
await 3 anim=run
await 3 anim=wait
send DBG puppet animlog
expect ANIMLOG.*\|\stest\s.*\srun\b
send DBG puppet drop
